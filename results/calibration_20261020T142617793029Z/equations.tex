\section*{Dynamics}
\subsection*{Compact form}
\subsubsection*{Bin Transitions}
\noindent\texttt{susceptble\_to\_latent (S -\textgreater{} E)}
\begin{align*}
S_{y65} \to E_{y65} &: \frac{\beta \cdot S_{y65} \cdot (IR + IH)}{total} \\
S_{o65,nv} \to E_{o65,nv} &: \frac{\beta \cdot S_{o65,nv} \cdot (IR + IH)}{total} \\
S_{o65,v} \to E_{o65,v} &: \frac{\beta \cdot S_{o65,v} \cdot (IR + IH)}{total}
\end{align*}
\noindent\texttt{exposed\_to\_infectious (E -\textgreater{} IR)}
\begin{align*}
E_{y65} \to IR_{y65} &: (1 - p_{h,y}) \cdot t_{latency} \cdot E_{y65} \\
E_{o65,nv} \to IR_{o65,nv} &: (1 - p_{h,nv}) \cdot t_{latency} \cdot E_{o65,nv} \\
E_{o65,v} \to IR_{o65,v} &: (1 - p_{h,v}) \cdot t_{latency} \cdot E_{o65,v}
\end{align*}
\noindent\texttt{exposed\_to\_hospital\_bound\_infectious (E -\textgreater{} IH)}
\begin{align*}
E_{y65} \to IH_{y65} &: p_{h,y} \cdot t_{latency} \cdot E_{y65} \\
E_{o65,nv} \to IH_{o65,nv} &: p_{h,nv} \cdot t_{latency} \cdot E_{o65,nv} \\
E_{o65,v} \to IH_{o65,v} &: p_{h,v} \cdot t_{latency} \cdot E_{o65,v}
\end{align*}
\noindent\texttt{infectious\_to\_recovered (IR -\textgreater{} R)}
\begin{align*}
IR_{y65} \to R_{y65} &: t_{recovery} \cdot IR_{y65} \\
IR_{o65,nv} \to R_{o65,nv} &: t_{recovery} \cdot IR_{o65,nv} \\
IR_{o65,v} \to R_{o65,v} &: t_{recovery} \cdot IR_{o65,v}
\end{align*}
\noindent\texttt{hospital\_bound\_infectious\_to\_hospitalized (IH -\textgreater{} H)}
\begin{align*}
IH_{y65} \to H_{y65} &: t_{hospitalization,nv} \cdot IH_{y65} \\
IH_{o65,nv} \to H_{o65,nv} &: t_{hospitalization,nv} \cdot IH_{o65,nv} \\
IH_{o65,v} \to H_{o65,v} &: t_{hospitalization,v} \cdot IH_{o65,v}
\end{align*}
\noindent\texttt{hospitalized\_to\_recovered (H -\textgreater{} R)}
\begin{align*}
H_{y65} \to R_{y65} &: t_{hospitalization,recovery,nv} \cdot H_{y65} \\
H_{o65,nv} \to R_{o65,nv} &: t_{hospitalization,recovery,nv} \cdot H_{o65,nv} \\
H_{o65,v} \to R_{o65,v} &: t_{hospitalization,recovery,v} \cdot H_{o65,v}
\end{align*}
\noindent\texttt{vaccination\_S (S -\textgreater{} S)}
\begin{align*}
S_{o65,nv} \to S_{o65,v} &: \frac{doses \cdot S_{o65,nv}}{S_{o65,nv} + R_{o65,nv}}
\end{align*}
\noindent\texttt{vaccination\_R (R -\textgreater{} R)}
\begin{align*}
R_{o65,nv} \to R_{o65,v} &: \frac{doses \cdot R_{o65,nv}}{S_{o65,nv} + R_{o65,nv}}
\end{align*}
\noindent\texttt{births (none -\textgreater{} S)}
\begin{align*}
\varnothing \to S_{y65} &: births
\end{align*}
\noindent\texttt{S\_y65\_to\_S\_o65 (S -\textgreater{} S)}
\begin{align*}
S_{y65} \to S_{o65,nv} &: aging \cdot S_{y65}
\end{align*}
\noindent\texttt{E\_y65\_to\_E\_o65 (E -\textgreater{} E)}
\begin{align*}
E_{y65} \to E_{o65,nv} &: aging \cdot E_{y65}
\end{align*}
\noindent\texttt{IR\_y65\_to\_IR\_o65 (IR -\textgreater{} IR)}
\begin{align*}
IR_{y65} \to IR_{o65,nv} &: aging \cdot IR_{y65}
\end{align*}
\noindent\texttt{IH\_y65\_to\_IH\_o65 (IH -\textgreater{} IH)}
\begin{align*}
IH_{y65} \to IH_{o65,nv} &: aging \cdot IH_{y65}
\end{align*}
\noindent\texttt{R\_y65\_to\_R\_o65 (R -\textgreater{} R)}
\begin{align*}
R_{y65} \to R_{o65,nv} &: aging \cdot R_{y65}
\end{align*}
\noindent\texttt{S\_death (S -\textgreater{} none)}
\begin{align*}
S_{y65} \to \varnothing &: death_{y65} \cdot S_{y65} \\
S_{o65,nv} \to \varnothing &: death_{o65} \cdot S_{o65,nv} \\
S_{o65,v} \to \varnothing &: death_{o65} \cdot S_{o65,v}
\end{align*}
\noindent\texttt{E\_death (E -\textgreater{} none)}
\begin{align*}
E_{y65} \to \varnothing &: death_{y65} \cdot E_{y65} \\
E_{o65,nv} \to \varnothing &: death_{o65} \cdot E_{o65,nv} \\
E_{o65,v} \to \varnothing &: death_{o65} \cdot E_{o65,v}
\end{align*}
\noindent\texttt{IR\_death (IR -\textgreater{} none)}
\begin{align*}
IR_{y65} \to \varnothing &: death_{y65} \cdot IR_{y65} \\
IR_{o65,nv} \to \varnothing &: death_{o65} \cdot IR_{o65,nv} \\

\end{align*}
\noindent\texttt{IH\_death (IH -\textgreater{} none)}

IH_{o65,nv} \to \varnothing &: death_{o65} \cdot IH_{o65,nv} \\
IH_{o65,v} \to \varnothing &: death_{o65} \cdot IH_{o65,v}

\noindent\texttt{H\_death (H -\textgreater{} none)}

H_{o65,nv} \to \varnothing &: death_{o65} \cdot H_{o65,nv} \\
H_{o65,v} \to \varnothing &: death_{o65} \cdot H_{o65,v}
\end{align*}
\noindent\texttt{R\_death (R -\textgreater{} none)}
\begin{align*}
R_{y65} \to \varnothing &: death_{y65} \cdot R_{y65} \\
R_{o65,nv} \to \varnothing &: death_{o65} \cdot R_{o65,nv} \\

Total System: 18 coupled equations (6 bins $\times$ 2 age $\times$ 2 vaccination)

\subsection*{Expanded form}
\begin{align*}
\frac{S_{y65}(t+\Delta t) - S_{y65}(t)}{\Delta t} &= - (\frac{\beta \cdot S_{y65} \cdot (IR + IH)}{total}) + (births) - (aging \cdot S_{y65}) - (death_{y65} \cdot S_{y65}) \\
\frac{S_{o65,nv}(t+\Delta t) - S_{o65,nv}(t)}{\Delta t} &= - (\frac{\beta \cdot S_{o65,nv} \cdot (IR + IH)}{total}) - (\frac{doses \cdot S_{o65,nv}}{S_{o65,nv} + R_{o65,nv}}) + (aging \cdot S_{y65}) - (death_{o65} \cdot S_{o65,nv}) \\
\frac{S_{o65,v}(t+\Delta t) - S_{o65,v}(t)}{\Delta t} &= - (\frac{\beta \cdot S_{o65,v} \cdot (IR + IH)}{total}) + (\frac{doses \cdot S_{o65,nv}}{S_{o65,nv} + R_{o65,nv}}) - (death_{o65} \cdot S_{o65,v}) \\
\frac{E_{y65}(t+\Delta t) - E_{y65}(t)}{\Delta t} &= (\frac{\beta \cdot S_{y65} \cdot (IR + IH)}{total}) - (t_{latency} \cdot E_{y65}) - (aging \cdot E_{y65}) - (death_{y65} \cdot E_{y65}) \\
\frac{E_{o65,nv}(t+\Delta t) - E_{o65,nv}(t)}{\Delta t} &= (\frac{\beta \cdot S_{o65,nv} \cdot (IR + IH)}{total}) - (t_{latency} \cdot E_{o65,nv}) + (aging \cdot E_{y65}) - (death_{o65} \cdot E_{o65,nv}) \\
\frac{E_{o65,v}(t+\Delta t) - E_{o65,v}(t)}{\Delta t} &= (\frac{\beta \cdot S_{o65,v} \cdot (IR + IH)}{total}) - (t_{latency} \cdot E_{o65,v}) - (death_{o65} \cdot E_{o65,v}) \\
\frac{IR_{y65}(t+\Delta t) - IR_{y65}(t)}{\Delta t} &= ((1 - p_{h,y}) \cdot t_{latency} \cdot E_{y65}) - (t_{recovery} \cdot IR_{y65}) - (aging \cdot IR_{y65}) - (death_{y65} \cdot IR_{y65}) \\
\frac{IR_{o65,nv}(t+\Delta t) - IR_{o65,nv}(t)}{\Delta t} &= ((1 - p_{h,nv}) \cdot t_{latency} \cdot E_{o65,nv}) - (t_{recovery} \cdot IR_{o65,nv}) + (aging \cdot IR_{y65}) - (death_{o65} \cdot IR_{o65,nv}) \\
\frac{IR_{o65,v}(t+\Delta t) - IR_{o65,v}(t)}{\Delta t} &= ((1 - p_{h,v}) \cdot t_{latency} \cdot E_{o65,v}) - (t_{recovery} \cdot IR_{o65,v}) - (death_{o65} \cdot IR_{o65,v}) \\
\frac{IH_{y65}(t+\Delta t) - IH_{y65}(t)}{\Delta t} &= (p_{h,y} \cdot t_{latency} \cdot E_{y65}) - (t_{hospitalization,nv} \cdot IH_{y65}) - (aging \cdot IH_{y65}) - (death_{y65} \cdot IH_{y65}) \\
\frac{IH_{o65,nv}(t+\Delta t) - IH_{o65,nv}(t)}{\Delta t} &= (p_{h,nv} \cdot t_{latency} \cdot E_{o65,nv}) - (t_{hospitalization,nv} \cdot IH_{o65,nv}) + (aging \cdot IH_{y65}) - (death_{o65} \cdot IH_{o65,nv}) \\
\frac{IH_{o65,v}(t+\Delta t) - IH_{o65,v}(t)}{\Delta t} &= (p_{h,v} \cdot t_{latency} \cdot E_{o65,v}) - (t_{hospitalization,v} \cdot IH_{o65,v}) - (death_{o65} \cdot IH_{o65,v}) \\
\frac{H_{y65}(t+\Delta t) - H_{y65}(t)}{\Delta t} &= (t_{hospitalization,nv} \cdot IH_{y65}) - (t_{hospitalization,recovery,nv} \cdot H_{y65}) - (death_{y65} \cdot H_{y65}) \\
\frac{H_{o65,nv}(t+\Delta t) - H_{o65,nv}(t)}{\Delta t} &= (t_{hospitalization,nv} \cdot IH_{o65,nv}) - (t_{hospitalization,recovery,nv} \cdot H_{o65,nv}) - (death_{o65} \cdot H_{o65,nv}) \\
\frac{H_{o65,v}(t+\Delta t) - H_{o65,v}(t)}{\Delta t} &= (t_{hospitalization,v} \cdot IH_{o65,v}) - (t_{hospitalization,recovery,v} \cdot H_{o65,v}) - (death_{o65} \cdot H_{o65,v}) \\
\frac{R_{y65}(t+\Delta t) - R_{y65}(t)}{\Delta t} &= (t_{recovery} \cdot IR_{y65}) + (t_{hospitalization,recovery,nv} \cdot H_{y65}) - (aging \cdot R_{y65}) - (death_{y65} \cdot R_{y65}) \\
\frac{R_{o65,nv}(t+\Delta t) - R_{o65,nv}(t)}{\Delta t} &= (t_{recovery} \cdot IR_{o65,nv}) + (t_{hospitalization,recovery,nv} \cdot H_{o65,nv}) - (\frac{doses \cdot R_{o65,nv}}{S_{o65,nv} + R_{o65,nv}}) + (aging \cdot R_{y65}) - (death_{o65} \cdot R_{o65,nv}) \\
\frac{R_{o65,v}(t+\Delta t) - R_{o65,v}(t)}{\Delta t} &= (t_{recovery} \cdot IR_{o65,v}) + (t_{hospitalization,recovery,v} \cdot H_{o65,v}) + (\frac{doses \cdot R_{o65,nv}}{S_{o65,nv} + R_{o65,nv}}) - (death_{o65} \cdot R_{o65,v})
\end{align*}
\section*{Schedule}
Each step applies every flow once, computed from the state at the start of the step: X(n + 1) = X(n) + sum of the flows. The impulses of a step act first, and its flows run from the new state.
